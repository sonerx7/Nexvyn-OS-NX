cmake_minimum_required(VERSION 3.13)
project(io_demo LANGUAGES C CXX)
if(NOT DEFINED NEXVYN_KIT)
  message(FATAL_ERROR "Build with the exported kit's nxv_native tool or provide NEXVYN_KIT")
endif()
set(NEXVYN_ABI ON CACHE BOOL "" FORCE)
include(${NEXVYN_KIT}/toolchain/nexvyn-native.cmake)
add_executable(io_demo main.c)
target_compile_options(io_demo PRIVATE -Wall -Wextra -Werror)
target_include_directories(io_demo SYSTEM PRIVATE "${DEVKITPRO}/portlibs/switch/include")
nexvyn_add_nro(io_demo TITLE "Input and audio demo" AUTHOR "Nexvyn Team"
               VERSION "1.0.0" ICON "${CMAKE_CURRENT_SOURCE_DIR}/icon.png")
if(NEXVYN_WINDOWED)
  target_compile_options(io_demo_windowed_objects PRIVATE -Wall -Wextra -Werror)
else()
  # The standalone implementation of nxn_audio and nxn_display uses these backends.
  find_package(PkgConfig REQUIRED)
  pkg_check_modules(N6_AUDIO REQUIRED IMPORTED_TARGET sdl2)
  target_link_libraries(io_demo PRIVATE PkgConfig::N6_AUDIO)
endif()
